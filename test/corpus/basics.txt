func classify(n: int) -> str {
    if n > 0 {
        return "pos"
    } else {
        return "neg"
    }
    match n {
        .some(v) => v,
        _ => 0,
    }
    for x in 0..10 {
        println("{x}")
    }
    while true {
        break
    }
    return "done"
}
